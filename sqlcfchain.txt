-- Bảng Roles (Vai trò)

-- Bảng Stores (Cửa hàng)
IF OBJECT_ID(N'dbo.Stores', N'U') IS NULL
BEGIN
    CREATE TABLE dbo.Stores (
        StoreId INT IDENTITY(1,1) NOT NULL CONSTRAINT PK_Stores PRIMARY KEY,
        StoreName NVARCHAR(100) NOT NULL,
        Address NVARCHAR(255) NOT NULL,
        ManagerId INT NULL,
        TotalLeaveDays INT NOT NULL CONSTRAINT DF_Stores_TotalLeaveDays DEFAULT (12),
        IsActive BIT NOT NULL CONSTRAINT DF_Stores_IsActive DEFAULT (1),
        CreatedAt DATETIME2 NOT NULL CONSTRAINT DF_Stores_CreatedAt DEFAULT (SYSUTCDATETIME())
    );
    PRINT N'Đã tạo bảng Stores.';
END
GO

-- Bảng StoreOperatingHours (Giờ mở cửa)
IF OBJECT_ID(N'dbo.StoreOperatingHours', N'U') IS NULL
BEGIN
    CREATE TABLE dbo.StoreOperatingHours (
        OperatingHourId INT IDENTITY(1,1) NOT NULL CONSTRAINT PK_StoreOperatingHours PRIMARY KEY,
        StoreId INT NOT NULL,
        DayOfWeek TINYINT NOT NULL,
        OpenTime TIME NULL,
        CloseTime TIME NULL,
        IsOpen BIT NOT NULL CONSTRAINT DF_StoreOperatingHours_IsOpen DEFAULT (1),
        CONSTRAINT FK_StoreOperatingHours_Store FOREIGN KEY (StoreId) REFERENCES dbo.Stores(StoreId) ON DELETE CASCADE,
        CONSTRAINT UQ_StoreOperatingHours_Store_Day UNIQUE (StoreId, DayOfWeek),
        CONSTRAINT CK_StoreOperatingHours_DayOfWeek CHECK (DayOfWeek BETWEEN 0 AND 6)
    );
    PRINT N'Đã tạo bảng StoreOperatingHours.';
END
GO

-- Bảng Employees (Nhân viên)
IF OBJECT_ID(N'dbo.Employees', N'U') IS NULL
BEGIN
    CREATE TABLE dbo.Employees (
        EmployeeId INT IDENTITY(1,1) NOT NULL CONSTRAINT PK_Employees PRIMARY KEY,
        FullName NVARCHAR(100) NOT NULL,
Phone VARCHAR(20) NOT NULL,
        Email VARCHAR(100) NOT NULL CONSTRAINT UQ_Employees_Email UNIQUE,
        Address NVARCHAR(255) NULL,
        StoreId INT NOT NULL,
        Status NVARCHAR(20) NOT NULL CONSTRAINT DF_Employees_Status DEFAULT (N'Active'),
        HireDate DATE NOT NULL,
        TerminationDate DATE NULL,

END
GO

-- Bảng ShiftAssignments (Phân công ca làm việc)
IF OBJECT_ID(N'dbo.ShiftAssignments', N'U') IS NULL
BEGIN
    CREATE TABLE dbo.ShiftAssignments (
        AssignmentId INT IDENTITY(1,1) NOT NULL CONSTRAINT PK_ShiftAssignments PRIMARY KEY,
        ShiftId INT NOT NULL,
        EmployeeId INT NOT NULL,
        WorkDate DATE NOT NULL,
        Status NVARCHAR(20) NOT NULL CONSTRAINT DF_ShiftAssignments_Status DEFAULT (N'Assigned'),

        CONSTRAINT FK_ShiftAssignments_Employee FOREIGN KEY (EmployeeId) REFERENCES dbo.Employees(EmployeeId),
        CONSTRAINT UQ_Assignment_EmpShiftDate UNIQUE (EmployeeId, ShiftId, WorkDate)

    CREATE INDEX IX_ShiftAssignments_Store ON dbo.ShiftAssignments(ShiftId, WorkDate);
    CREATE INDEX IX_ShiftAssignments_Date ON dbo.ShiftAssignments(WorkDate);

END

    SELECT 1 FROM sys.columns

    ALTER TABLE dbo.ShiftChangeRequests

END
GO

-- Cột TargetEmployeeId cho ShiftChangeRequests (Chuyển ca 1 chiều)

PRINT N'Đã thêm cột TargetEmployeeId vào ShiftChangeRequests.';
END
GO

        NotificationId INT IDENTITY(1,1) NOT NULL CONSTRAINT PK_Notifications PRIMARY KEY,
        UserId INT NOT NULL,
        Title NVARCHAR(200) NOT NULL,
        Message NVARCHAR(1000) NOT NULL,
        NotificationType VARCHAR(50) NOT NULL,
        IsRead BIT NOT NULL CONSTRAINT DF_Notifications_IsRead DEFAULT (0),
        ReferenceType VARCHAR(50) NULL,
        ReferenceId INT NULL,
        CreatedAt DATETIME2 NOT NULL CONSTRAINT DF_Notifications_CreatedAt DEFAULT (SYSUTCDATETIME()),
        CONSTRAINT FK_Notifications_User FOREIGN KEY (UserId) REFERENCES dbo.Users (UserId)
    );

    CREATE INDEX IX_Notifications_User_Read_CreatedAt
        ON dbo.Notifications (UserId, IsRead, CreatedAt DESC);
END;
GO
